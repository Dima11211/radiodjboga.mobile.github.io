// В начале файла function getStreamUrl() { return "https://radio.radiodjboga.online/stream?_=" + Date.now(); } async function startRadio() { if (!navigator.onLine) { showOfflineOverlay(); return; } try { // Полная очистка старого if (audio) { audio.pause(); audio.src = ''; audio.load(); audio = null; } if (ctxAudio && ctxAudio.state !== 'closed') { if (animation) cancelAnimationFrame(animation); try { await ctxAudio.close(); } catch(e) {} ctxAudio = null; analyser = null; } const newAudio = new Audio(); newAudio.crossOrigin = 'anonymous'; newAudio.src = getStreamUrl(); // Случайный параметр newAudio.preload = 'none'; // Не кэшировать await new Promise((resolve, reject) => { const tm = setTimeout(() => reject('Таймаут'), 15000); newAudio.addEventListener('canplay', () => { clearTimeout(tm); resolve(); }, { once: true }); newAudio.addEventListener('error', (e) => { clearTimeout(tm); reject('Ошибка: ' + e.target.error?.message); }, { once: true }); newAudio.load(); }); audio = newAudio; const ok = await initAudio(); if (!ok) throw new Error('Ошибка звука'); if (ctxAudio && ctxAudio.state === 'suspended') await ctxAudio.resume(); await audio.play(); isPlaying = true; playBtn.innerHTML = '⏹ STOP'; playBtn.classList.add('playing'); } catch (err) { console.error('Ошибка:', err); isPlaying = false; playBtn.innerHTML = '▶ PLAY'; playBtn.classList.remove('playing'); if (audio) { audio.pause(); audio.src = ''; audio.load(); audio = null; } } } function stopRadio() { if (audio) { audio.pause(); audio.currentTime = 0; audio.src = ''; audio.load(); audio = null; } isPlaying = false; playBtn.innerHTML = '▶ PLAY'; playBtn.classList.remove('playing'); if (ctxAudio && ctxAudio.state !== 'closed') { if (animation) cancelAnimationFrame(animation); try { ctxAudio.close(); } catch(e) {} ctxAudio = null; analyser = null; } }
